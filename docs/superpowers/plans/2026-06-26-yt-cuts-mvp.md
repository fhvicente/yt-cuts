# yt-cuts MVP (Fase 1) Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** CLI que recebe uma URL do YouTube e produz N clips verticais 9:16 com legendas karaoke queimadas, sem intervenção manual.

**Architecture:** Pipeline sequencial com stages plugáveis (strategy pattern). Inteligência (ASR=Deepgram, seleção de cortes=Claude) via API pay-per-use; média (download, crop, burn-in, encode) local em CPU via FFmpeg. Cache de artefactos por video-id evita re-pagar ASR em re-runs.

**Tech Stack:** Python 3.12, yt-dlp, deepgram-sdk (v3), anthropic SDK, FFmpeg/libass (binário de sistema), python-dotenv, pytest.

**Spec:** `docs/superpowers/specs/2026-06-26-gerador-cortes-design.md`

## Global Constraints

- **Sem git este sessão** — utilizador optou por não fazer commits nem worktree. Cada task termina em testes verdes (não em commit). Inicializar git + commitar depois se quiser.
- **Zero chamadas API live em testes.** ASR/LLM mockados; ffmpeg testado contra vídeo gerado por `ffmpeg lavfi` em runtime (sem binários no repo).
- **Python ≥ 3.12.** Type hints modernos (`X | None`).
- **Instalar deps no `.venv` existente:** `.venv/bin/pip install ...`. Correr testes com `.venv/bin/pytest`.
- **Modelo Claude default:** `claude-sonnet-4-6` (equilíbrio custo/qualidade p/ seleção de cortes; ~$0.03 por transcrição de 1h). Configurável.
- **Modelo Deepgram default:** `nova-3`.
- **Dimensões de output fixas:** 1080×1920 (9:16). Dims pares (libx264).
- **Legendas karaoke .ass:** `PlayResX 1080 / PlayResY 1920`; PrimaryColour = cor ativa (amarelo), SecondaryColour = inativa (branco); sweep via `\kf`.

`yt_cuts/cache.py`:
```python
from __future__ import annotations

import json
from dataclasses import asdict
from pathlib import Path

from yt_cuts.models import Transcript, Word


class Cache:
    def __init__(self, root: Path, video_id: str, enabled: bool = True):
        self.dir = Path(root) / video_id
        self.enabled = enabled

    def path(self, name: str) -> Path:
        return self.dir / name

    def has(self, name: str) -> bool:
        return self.enabled and self.path(name).exists()

    def ensure_dir(self) -> None:
        self.dir.mkdir(parents=True, exist_ok=True)

    def save_transcript(self, t: Transcript) -> None:
        self.ensure_dir()
        payload = {
            "language": t.language,
            "duration": t.duration,
            "words": [asdict(w) for w in t.words],
        }
        self.path("transcript.json").write_text(json.dumps(payload, ensure_ascii=False))

    def load_transcript(self) -> Transcript:
        data = json.loads(self.path("transcript.json").read_text())
        words = [Word(**w) for w in data["words"]]
        return Transcript(words=words, language=data["language"], duration=data["duration"])
```

```python
from __future__ import annotations

from pathlib import Path

TARGET_W, TARGET_H = 1080, 1920
TARGET_RATIO = TARGET_W / TARGET_H  # 0.5625


def _even(n: int) -> int:
    return n - (n % 2)


class StaticCenterReframer:
    def crop_filter(self, src_w: int, src_h: int, clip_path: Path | None = None) -> str:
        crop_w = _even(min(src_w, round(src_h * TARGET_RATIO)))
        crop_h = _even(min(src_h, round(crop_w / TARGET_RATIO)))
        x = (src_w - crop_w) // 2
        y = (src_h - crop_h) // 2
        return f"crop={crop_w}:{crop_h}:{x}:{y},scale={TARGET_W}:{TARGET_H}"
```

`yt_cuts/subtitles/ass_builder.py`:
```python
from __future__ import annotations

from yt_cuts.models import Segment, Word
from yt_cuts.subtitles.styles import Style, style_for

MAX_WORDS_PER_LINE = 5
MAX_GAP = 0.8  # segundos: gap maior quebra linha


def fmt_time(t: float) -> str:
    cs = int(round(t * 100))
    h, cs = divmod(cs, 360000)
    m, cs = divmod(cs, 6000)
    s, cs = divmod(cs, 100)
    return f"{h}:{m:02d}:{s:02d}.{cs:02d}"


def _header(s: Style) -> str:
    return (
        "[Script Info]\n"
        "ScriptType: v4.00+\n"
        "PlayResX: 1080\n"
        "PlayResY: 1920\n"
        "WrapStyle: 2\n\n"
        "[V4+ Styles]\n"
        "Format: Name, Fontname, Fontsize, PrimaryColour, SecondaryColour, "
        "OutlineColour, BackColour, Bold, Italic, Underline, StrikeOut, ScaleX, "
        "ScaleY, Spacing, Angle, BorderStyle, Outline, Shadow, Alignment, "
        "MarginL, MarginR, MarginV, Encoding\n"
        f"Style: Karaoke,{s.font},{s.size},{s.primary},{s.secondary},"
        f"{s.outline_colour},{s.back_colour},{s.bold},0,0,0,100,100,0,0,1,"
        f"{s.outline},{s.shadow},{s.alignment},60,60,{s.margin_v},1\n\n"
        "[Events]\n"
        "Format: Layer, Start, End, Style, Name, MarginL, MarginR, MarginV, "
        "Effect, Text\n"
    )


def _group_lines(words: list[Word]) -> list[list[Word]]:
    lines: list[list[Word]] = []
    cur: list[Word] = []
    for w in words:
        if cur and (len(cur) >= MAX_WORDS_PER_LINE or w.start - cur[-1].end > MAX_GAP):
            lines.append(cur)
            cur = []
        cur.append(w)
    if cur:
        lines.append(cur)
    return lines


def build_ass(words: list[Word], segment: Segment, style: Style | None = None) -> str:
    style = style or style_for("karaoke")
    rebased = [
        Word(w.text, max(0.0, w.start - segment.start), max(0.0, w.end - segment.start))
        for w in words
        if w.end > segment.start and w.start < segment.end
    ]
    events: list[str] = []
    for line in _group_lines(rebased):
        start = fmt_time(line[0].start)
        end = fmt_time(line[-1].end)
        text = "".join(
            f"{{\\kf{max(1, int(round((w.end - w.start) * 100)))}}}{w.text} "
            for w in line
        ).strip()
        events.append(f"Dialogue: 0,{start},{end},Karaoke,,0,0,0,,{text}")
```

```python
from __future__ import annotations

from yt_cuts.models import Segment, Word


def snap_to_words(seg: Segment, words: list[Word]) -> Segment:
    if not words:
        return seg
    start = min((w.start for w in words), key=lambda s: abs(s - seg.start))
    end = min((w.end for w in words), key=lambda e: abs(e - seg.end))
    return Segment(start, end, seg.title, seg.hook, seg.score)


def clamp_duration(seg: Segment, min_dur: float, max_dur: float) -> Segment | None:
    if seg.duration < min_dur:
        return None
    if seg.duration > max_dur:
        return Segment(seg.start, seg.start + max_dur, seg.title, seg.hook, seg.score)
    return seg


def dedupe_overlaps(segs: list[Segment]) -> list[Segment]:
    ordered = sorted(segs, key=lambda s: (-(s.score or 0.0), s.start))
    kept: list[Segment] = []
    for s in ordered:
        if any(s.start < k.end and k.start < s.end for k in kept):
            continue
        kept.append(s)
    return kept


def finalize(
    raw: list[Segment], words: list[Word], n: int, min_dur: float, max_dur: float
) -> list[Segment]:
    snapped = [snap_to_words(s, words) for s in raw]
    clamped = [c for c in (clamp_duration(s, min_dur, max_dur) for s in snapped) if c]
    deduped = dedupe_overlaps(clamped)
    deduped.sort(key=lambda s: -(s.score or 0.0))
    return deduped[:n]
```

```python
from __future__ import annotations

import json
import subprocess
from pathlib import Path


class FFmpegError(RuntimeError):
    pass


def run(args: list[str], *, check: bool = True) -> subprocess.CompletedProcess:
    cmd = ["ffmpeg", "-y", "-hide_banner", "-loglevel", "error", *args]
    proc = subprocess.run(cmd, capture_output=True, text=True)
    if check and proc.returncode != 0:
        raise FFmpegError(proc.stderr.strip() or f"ffmpeg falhou: {' '.join(cmd)}")
    return proc


def probe_dimensions(path: Path | str) -> tuple[int, int]:
    cmd = [
        "ffprobe", "-v", "error", "-select_streams", "v:0",
        "-show_entries", "stream=width,height", "-of", "json", str(path),
    ]
    proc = subprocess.run(cmd, capture_output=True, text=True)
    if proc.returncode != 0:
        raise FFmpegError(proc.stderr.strip())
    stream = json.loads(proc.stdout)["streams"][0]
    return int(stream["width"]), int(stream["height"])
```

```python
from pathlib import Path
```

```python
class DownloadError(RuntimeError):
    pass


def probe_video(url: str) -> tuple[str, dict]:
    try:
        with yt_dlp.YoutubeDL({"quiet": True, "skip_download": True}) as ydl:
            info = ydl.extract_info(url, download=False)
    except Exception as e:  # noqa: BLE001
        raise DownloadError(f"yt-dlp falhou a ler {url}. Atualiza: pip install -U yt-dlp") from e

def download(url: str, dest_dir: Path, video_id: str) -> Path:

        "merge_output_format": "mp4",
        "outtmpl": str(dest_dir / "%(id)s.%(ext)s"),
        "quiet": True,
    }
    try:
        with yt_dlp.YoutubeDL(opts) as ydl:
            ydl.extract_info(url, download=True)
    except Exception as e:  # noqa: BLE001
        raise DownloadError("yt-dlp falhou o download. Atualiza: pip install -U yt-dlp") from e
    return dest_dir / f"{video_id}.mp4"
```

```python
from yt_cuts.models import Transcript, Word


def map_response(data: dict, language: str = "") -> Transcript:
    alt = data["results"]["channels"][0]["alternatives"][0]
    words = [
        Word(
            text=w["word"],

            diarize=True,
            punctuate=True,
```

```python
from __future__ import annotations

from pathlib import Path

from yt_cuts.ffmpeg_utils import run
from yt_cuts.models import ClipResult, Segment

OUT_W, OUT_H = 1080, 1920


def _escape_ass_path(path: Path) -> str:
    return str(path).replace("\\", "\\\\").replace(":", "\\:")


def render_clip(
    video_path: Path,
    segment: Segment,
    ass_path: Path,
    out_path: Path,
    reframer,
    src_dims: tuple[int, int],
) -> ClipResult:
    src_w, src_h = src_dims
    vf = f"{reframer.crop_filter(src_w, src_h)},ass={_escape_ass_path(ass_path)}"
    run([
        "-ss", f"{segment.start}", "-to", f"{segment.end}", "-i", str(video_path),
        "-vf", vf,
        "-c:v", "libx264", "-preset", "medium", "-crf", "20",
        "-c:a", "aac", "-b:a", "128k", "-movflags", "+faststart",
        str(out_path),
    ])
    return ClipResult(segment=segment, path=out_path, width=OUT_W, height=OUT_H)
```

`yt_cuts/pipeline.py`:
```python
from __future__ import annotations

from yt_cuts.asr.base import get_asr
from yt_cuts.audio import extract_audio
from yt_cuts.cache import Cache
from yt_cuts.config import Settings
from yt_cuts.cuts.base import get_selector
from yt_cuts.download import download, probe_video
from yt_cuts.ffmpeg_utils import probe_dimensions
from yt_cuts.models import ClipResult
from yt_cuts.reframe.base import get_reframer
from yt_cuts.render import render_clip
from yt_cuts.subtitles.ass_builder import build_ass
from yt_cuts.subtitles.styles import style_for


def run(url: str, settings: Settings) -> list[ClipResult]:
    settings.out.mkdir(parents=True, exist_ok=True)
    asr = get_asr(settings.asr, settings)
    selector = get_selector(settings.llm, settings)
    reframer = get_reframer(settings.reframe)
    style = style_for(settings.style)

    video_id, _info = probe_video(url)
    cache = Cache(settings.cache_dir, video_id, settings.use_cache)

    video = cache.path("video.mp4")
    if not cache.has("video.mp4"):
        cache.ensure_dir()
        video = download(url, cache.dir, video_id)

    audio = cache.path("audio.wav")
    if not cache.has("audio.wav"):
        extract_audio(video, audio)

    if cache.has("transcript.json"):
        transcript = cache.load_transcript()
    else:
        transcript = asr.transcribe(audio, settings.idioma)
        cache.save_transcript(transcript)

    segments = selector.select(transcript, settings.clips, settings.min_dur, settings.max_dur)
    src_dims = probe_dimensions(video)

    results: list[ClipResult] = []
    for i, seg in enumerate(segments):
        words = [w for w in transcript.words if w.end > seg.start and w.start < seg.end]
        ass_path = settings.out / f"clip_{i:02d}.ass"
        ass_path.write_text(build_ass(words, seg, style), encoding="utf-8")
        out_path = settings.out / f"clip_{i:02d}.mp4"
        results.append(render_clip(video, seg, ass_path, out_path, reframer, src_dims))
    return results
```

```python
from __future__ import annotations

import argparse
from pathlib import Path

from yt_cuts import pipeline
from yt_cuts.config import Settings


def main(argv: list[str] | None = None) -> int:
    p = argparse.ArgumentParser(prog="cortar.py", description="Gera cortes verticais legendados.")
    p.add_argument("url")
    p.add_argument("--clips", type=int, default=10)
    p.add_argument("--idioma", default="pt")
    p.add_argument("--style", default="karaoke")
    p.add_argument("--out", type=Path, default=Path("./output"))
    p.add_argument("--asr", default="deepgram")
    p.add_argument("--llm", default="claude")
    p.add_argument("--reframe", default="static", choices=["static"])
    p.add_argument("--no-cache", action="store_true")
    p.add_argument("--min-dur", type=float, default=20.0)
    p.add_argument("--max-dur", type=float, default=60.0)
    a = p.parse_args(argv)

    settings = Settings.from_env(
        clips=a.clips, idioma=a.idioma, style=a.style, out=a.out,
        asr=a.asr, llm=a.llm, reframe=a.reframe, use_cache=not a.no_cache,
        min_dur=a.min_dur, max_dur=a.max_dur,
    )
    results = pipeline.run(a.url, settings)
    print(f"{len(results)} clips → {a.out}")
    return 0
```
